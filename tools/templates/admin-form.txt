"use client";

import { zodResolver } from "@hookform/resolvers/zod";
import { useEffect } from "react";
import { useForm } from "react-hook-form";

import { {{MODULE_CAMEL}}Schema, type {{MODULE_UPPER}}FormValues } from "../validation";

type {{MODULE_UPPER}}FormProps = {
  initialData?: Partial<{{MODULE_UPPER}}FormValues>;
  onSubmit?: (data: {{MODULE_UPPER}}FormValues) => void | Promise<void>;
  submitLabel?: string;
  cancelLabel?: string;
  onCancel?: () => void;
  disabled?: boolean;
};

export default function Form({
  initialData,
  onSubmit,
  submitLabel = "Simpan",
  cancelLabel = "Batal",
  onCancel,
  disabled = false,
}: {{MODULE_UPPER}}FormProps) {
  const {
    register,
    handleSubmit,
    reset,
    formState: { errors, isSubmitting },
  } = useForm<{{MODULE_UPPER}}FormValues>({
    resolver: zodResolver({{MODULE_CAMEL}}Schema),
    defaultValues: { name: initialData?.name ?? "" },
  });

  useEffect(() => {
    reset({ name: initialData?.name ?? "" });
  }, [initialData, reset]);

  return (
    <form
      onSubmit={handleSubmit(async (data) => {
        await onSubmit?.(data);
      })}
      className="space-y-4"
    >
      <div>
        <label htmlFor="{{MODULE_KEBAB}}-name" className="block text-sm font-medium text-slate-700">
          Nama
        </label>

        <input
          id="{{MODULE_KEBAB}}-name"
          type="text"
          {...register("name")}
          className="mt-1 w-full rounded-md border border-slate-300 px-3 py-2"
          aria-invalid={Boolean(errors.name)}
          aria-describedby={errors.name ? "{{MODULE_KEBAB}}-name-error" : undefined}
        />

        {errors.name ? (
          <p id="{{MODULE_KEBAB}}-name-error" className="mt-1 text-sm text-red-600">
            {errors.name.message}
          </p>
        ) : null}
      </div>

      <div className="flex gap-2">
        <button
          type="submit"
          disabled={disabled || isSubmitting}
          className="rounded-md border border-slate-300 px-4 py-2"
        >
          {isSubmitting ? "Menyimpan..." : submitLabel}
        </button>

        {onCancel ? (
          <button
            type="button"
            disabled={disabled || isSubmitting}
            onClick={onCancel}
            className="rounded-md border border-slate-300 px-4 py-2"
          >
            {cancelLabel}
          </button>
        ) : null}
      </div>
    </form>
  );
}
